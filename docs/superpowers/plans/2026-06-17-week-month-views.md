# Week & Month Reading Views — Implementation Plan (Plan 1 of 2)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Spec:** [2026-06-17-week-month-reading-views-design.md](../specs/2026-06-17-week-month-reading-views-design.md)

**Goal:** Add a source-grouped, card-based **Week** and **Month** reading view (time-ordered) plus a one-tap **Hide** (not-interested) action with undo, so the user can review several days of unread at once without day-by-day switching and prune noise themselves.

**Architecture:** A new per-`(user, entry)` boolean `is_hidden` mirrors the existing `read_later` state end-to-end (model → schema → service → router → types → api-client → hook). A new front-end view (`view=week` / `view=month`) fetches a date *range* of **unread, non-hidden** entries via the existing paginated `/entries` endpoint (infinite scroll), groups them by source with the existing `buildTodayBoardGroups`, and renders the card layout from the approved mockup. Read items sink into a lazily-loaded per-source "看过的" box.

**Tech Stack:** Python 3.11 / FastAPI / SQLAlchemy async / Alembic / pytest; React 18 / TypeScript / TanStack Query / Vite / vitest.

**Scope note:** The `按兴趣` preference sort and its score/vector are **Plan 2** (they need the read/hide signals this plan starts generating — the spec's cold-start point). Plan 1 ships **time order only**; no sort control is rendered yet.

```ts
import {
  startOfWeek, endOfWeek, startOfMonth, endOfMonth,
  addWeeks, addMonths, format, parseISO,
} from 'date-fns'

export type PeriodKind = 'week' | 'month'

function localStartIso(d: Date): string {
  // Local-time start-of-day ISO without timezone suffix (matches existing range helpers)
  return `${format(d, 'yyyy-MM-dd')}T00:00:00.000`
}

export function getPeriodRange(anchorDateKey: string, period: PeriodKind) {
  const anchor = parseISO(anchorDateKey)
  const start = period === 'week'
    ? startOfWeek(anchor, { weekStartsOn: 1 })
    : startOfMonth(anchor)
  const endInclusive = period === 'week'
    ? endOfWeek(anchor, { weekStartsOn: 1 })
    : endOfMonth(anchor)
  const exclusiveEnd = new Date(endInclusive)
  exclusiveEnd.setHours(24, 0, 0, 0) // start of the next day
  return {
    collected_after: localStartIso(start),
    collected_before: localStartIso(exclusiveEnd),
    label: period === 'week'
      ? `${format(start, 'M月d日')} – ${format(endInclusive, 'M月d日')}`
      : format(anchor, 'yyyy年M月'),
  }
}

export function shiftPeriod(anchorDateKey: string, period: PeriodKind, delta: number): string {
  const anchor = parseISO(anchorDateKey)
  const next = period === 'week' ? addWeeks(anchor, delta) : addMonths(anchor, delta)
  return format(next, 'yyyy-MM-dd')
}
```

```ts
import { useMemo } from 'react'
import { useInfiniteQuery } from '@tanstack/react-query'
import { entryService } from '@glean/api-client'
import type { EntryWithState } from '@glean/types'

const PER_PAGE = 50

export function useRangeEntries(params: {
  collected_after: string
  collected_before: string
  enabled: boolean
}) {
  const { collected_after, collected_before, enabled } = params
  const query = useInfiniteQuery({
    queryKey: ['range-entries', collected_after, collected_before],
    queryFn: ({ pageParam = 1, signal }) =>
      entryService.getEntries(
        {
          collected_after,
          collected_before,
          is_read: false,
          is_hidden: false,
          page: pageParam,
          per_page: PER_PAGE,
        },
        { signal }
      ),
    getNextPageParam: (last) => (last.page < last.total_pages ? last.page + 1 : undefined),
    initialPageParam: 1,
    staleTime: 45 * 1000,
    gcTime: 30 * 60 * 1000,
    refetchOnWindowFocus: false,
    enabled,
  })
  const entries: EntryWithState[] = useMemo(
    () => query.data?.pages.flatMap((p) => p.items) ?? [],
    [query.data]
  )
  return { ...query, entries }
}
```

```ts
import { useCallback, useState } from 'react'
import type { TodayBoardEntry } from '../../todayBoard'

export function useHiddenUndo() {
  const [hidden, setHidden] = useState<Record<string, TodayBoardEntry[]>>({})
  const add = useCallback((e: TodayBoardEntry) => {
    setHidden((m) => ({ ...m, [e.feed_id]: [...(m[e.feed_id] ?? []), e] }))
  }, [])
  const remove = useCallback((e: TodayBoardEntry) => {
    setHidden((m) => ({ ...m, [e.feed_id]: (m[e.feed_id] ?? []).filter((x) => x.id !== e.id) }))
  }, [])
  return { hidden, add, remove }
}
```

```tsx
import { useEffect, useMemo, useRef } from 'react'
import { ChevronLeft, ChevronRight, Inbox } from 'lucide-react'
import { cn } from '@glean/ui'
import { useUpdateEntryState } from '@/hooks/useEntries'
import { useRangeEntries } from '@/hooks/useRangeEntries'
import { getPeriodRange, shiftPeriod, type PeriodKind } from '../../periodRange'
import { buildTodayBoardEntries, buildTodayBoardGroups, type TodayBoardEntry } from '../../todayBoard'
import { SourceSection } from './SourceSection'
import { useHiddenUndo } from './useHiddenUndo'

export function PeriodView({
  periodKind, anchorDate, onChangePeriod, onOpenEntry, renderDetail, selectedEntryId,
}: {
  periodKind: PeriodKind
  anchorDate: string
  onChangePeriod: (kind: 'day' | 'week' | 'month', anchor: string) => void
  onOpenEntry: (e: TodayBoardEntry) => void
  renderDetail?: (e: TodayBoardEntry) => React.ReactNode
  selectedEntryId: string | null
}) {
  const range = useMemo(() => getPeriodRange(anchorDate, periodKind), [anchorDate, periodKind])
  const q = useRangeEntries({ ...range, enabled: true })
  const update = useUpdateEntryState()
  const undo = useHiddenUndo()

  const groups = useMemo(() => {
    const built = buildTodayBoardEntries(q.entries, { filterBySelectedDate: false })
    const hiddenIds = new Set(Object.values(undo.hidden).flat().map((e) => e.id))
    return buildTodayBoardGroups(built.filter((e) => !hiddenIds.has(e.id)))
  }, [q.entries, undo.hidden])

  const hide = (e: TodayBoardEntry) => { undo.add(e); update.mutate({ entryId: e.id, data: { is_hidden: true } }) }
  const unhide = (e: TodayBoardEntry) => { undo.remove(e); update.mutate({ entryId: e.id, data: { is_hidden: false } }) }
  const toggleSave = (e: TodayBoardEntry) =>
    update.mutate({ entryId: e.id, data: { read_later: !e.read_later } })

  const sentinelRef = useRef<HTMLDivElement>(null)
  useEffect(() => {
    const el = sentinelRef.current
    if (!el || !q.hasNextPage) return
    const io = new IntersectionObserver((ents) => {
      if (ents[0]?.isIntersecting && !q.isFetchingNextPage) q.fetchNextPage()
    })
    io.observe(el)
    return () => io.disconnect()
  }, [q.hasNextPage, q.isFetchingNextPage, q.fetchNextPage])

  return (
    <div className="flex h-full min-w-0 flex-1 flex-col overflow-y-auto bg-[var(--color-background-secondary,#f7f5f0)]">
      <div className="sticky top-0 z-10 flex items-center justify-between gap-3 border-b border-border/60 bg-white/95 px-[18px] py-3 backdrop-blur">
        <div className="text-base font-medium text-stone-800">
          {periodKind === 'week' ? '本周' : '本月'}
          <span className="ml-2 text-xs font-normal text-stone-500">{range.label}</span>
        </div>
        <div className="flex items-center gap-2">
          <div className="inline-flex overflow-hidden rounded-md border border-border/60 text-xs">
            {(['day', 'week', 'month'] as const).map((k) => (
              <button key={k} type="button"
                onClick={() => onChangePeriod(k, anchorDate)}
                className={cn('px-2.5 py-1', k === periodKind ? 'bg-primary/10 text-primary font-medium' : 'text-stone-500')}>
                {k === 'day' ? '日' : k === 'week' ? '周' : '月'}
              </button>
            ))}
          </div>
          <button type="button" aria-label="上一期" onClick={() => onChangePeriod(periodKind, shiftPeriod(anchorDate, periodKind, -1))}>
            <ChevronLeft className="h-4 w-4 text-stone-500" />
          </button>
          <button type="button" aria-label="下一期" onClick={() => onChangePeriod(periodKind, shiftPeriod(anchorDate, periodKind, 1))}>
            <ChevronRight className="h-4 w-4 text-stone-500" />
          </button>
        </div>
      </div>

      {!q.isLoading && groups.length === 0 ? (
        <div className="flex flex-1 flex-col items-center justify-center py-20 text-center">
          <Inbox className="mb-3 h-8 w-8 text-stone-400" />
          <p className="text-sm text-stone-500">这段时间没有未读内容</p>
        </div>
      ) : (
        groups.map((g) => (
          <SourceSection key={g.feedId} group={g} range={range}
            justHidden={undo.hidden[g.feedId] ?? []}
            onOpen={onOpenEntry} onHide={hide} onUndo={unhide} onToggleSave={toggleSave} />
        ))
      )}
      <div ref={sentinelRef} className="h-8" />
    </div>
  )
}
```
